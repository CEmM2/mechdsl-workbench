% algorithm pcg
% backend taichi
% args A:matrix, b:vector, x:vector, apply_M_inv:callable, tol:scalar, maxiter:scalar

% type r vector
% type z vector
% type p vector
% type q vector
% type rho scalar
% type rho_new scalar
% type alpha scalar
% type beta scalar
% type pq scalar
% type r0_norm scalar
% type r_norm scalar

\begin{algorithmic}
\State $r = b - A \cdot x$
\State $r_0 = \lVert r \rVert_2$
\If{$r_0 = 0$}
    \Return $x, 0, 0$
\EndIf
\State $z = \text{apply\_M\_inv}(r)$
\State $p = z$
\State $\rho = r^\top z$
\For{$k = 1, 2, \ldots, \text{maxiter}$}
    \State $q = A \cdot p$
    \State $pq = p^\top q$
    \If{$|pq| < 10^{-300}$}
        \State \textbf{break}
    \EndIf
    \State $\alpha = \frac{\rho}{pq}$
    \State $x = x + \alpha \, p$
    \State $r = r - \alpha \, q$
    \State $r_n = \lVert r \rVert_2$
    \If{$r_n < \text{tol} \cdot r_0$}
        \Return $x, k, r_n$
    \EndIf
    \State $z = \text{apply\_M\_inv}(r)$
    \State $\rho_{\text{new}} = r^\top z$
    \State $\beta = \frac{\rho_{\text{new}}}{\rho}$
    \State $p = z + \beta \, p$
    \State $\rho = \rho_{\text{new}}$
\EndFor
\Return $x, \text{maxiter}, \lVert r \rVert_2$
\end{algorithmic}
